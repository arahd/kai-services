  sendMessageStream(payload: any): Observable<string> {
    return from(this.configService.loadConfig()).pipe(
      switchMap(() => {
        const url = `${this.configService.getChatbotUrl()}/chatservice`;

        const makeFetch = (token: string): Observable<string> => {
          const headers: Record<string, string> = {
            'Content-Type': 'application/json',
            'Authorization': `Bearer ${token}`,
          };

          return new Observable<string>((observer) => {
            fetch(url, { method: 'POST', headers, body: JSON.stringify(payload) })
              .then(async (response) => {
                // handle token expiry
                if (response.status === 401) {
                  console.warn('🔄 Stream token expired, refreshing...');
                  await this.refreshTokenAndRetry(async (newToken) => {
                    makeFetch(newToken).subscribe(observer);
                    return null as any;
                  });
                  return;
                }

                if (!response.ok || !response.body) {
                  throw new Error('Streaming request failed');
                }

                const reader = response.body.getReader();
                const decoder = new TextDecoder();

                const read = () => {
                  reader.read().then(({ done, value }) => {
                    if (done) {
                      observer.complete();
                      return;
                    }
                    const chunk = decoder.decode(value, { stream: true });
                    observer.next(chunk);
                    read();
                  });
                };

                read();
              })
              .catch((err) => observer.error(err));
          });
        };

        const existingToken = sessionStorage.getItem('jwt_token') || '';
        return makeFetch(existingToken);
      })
    );
  }

  private async postWithAuth<T>(url: string, body: any): Promise<T> {
    const token = sessionStorage.getItem('jwt_token');
    const headers = { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` };

    return response.json() as Promise<T>;

  /**
   * Helper method for GET with auth & token refresh
   */
  private async postGetWithAuth<T>(url: string, method: 'GET' | 'POST'): Promise<T> {
    const token = sessionStorage.getItem('jwt_token');
    const headers = { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` };

    const response = await fetch(url, { method, headers });

    if (response.status === 401) {
      return this.refreshTokenAndRetry(() => this.postGetWithAuth<T>(url, method));
    }

    if (!response.ok) throw new Error(`HTTP ${response.status}: ${response.statusText}`);
    return response.json() as Promise<T>;
  }

      switchMap(async () => {
        const url = `${this.configService.getAzureUrl()}/kai-reporting/get/user/all`;
        const token = sessionStorage.getItem('jwt_token') || '';
        const headers = { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` };
        const resp = await fetch(url, { method: 'POST', headers, body: JSON.stringify(payload) });
        if (resp.status === 401) {
          return this.refreshTokenAndRetry(async (newToken) => {
            const r = await fetch(url, { method: 'POST', headers: { ...headers, Authorization: `Bearer ${newToken}` }, body: JSON.stringify(payload) });
            return r.blob();
          });
        }
        if (!resp.ok) throw new Error('Export failed');
        return resp.blob();
      })
    );
  }
